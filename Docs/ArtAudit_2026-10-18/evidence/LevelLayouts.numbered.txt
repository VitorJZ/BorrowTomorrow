59:         /// Simple 模式下 size 不生效（灰盒曾全部渲染成贴图原尺寸小点）；

69:         /// <summary>灰盒盒子：SpriteRenderer + 可选 BoxCollider2D（可旋转）。</summary>
70:         internal static GameObject Box(RoomContext ctx, string name, Vector2 center, Vector2 size, Color color,
71:             string sortingLayer, string layerName, bool solid, bool oneWay = false, float angle = 0f)
72:         {
73:             var box = new GameObject(name);
74:             box.transform.SetParent(ctx.Content);
75:             box.transform.position = new Vector3(center.x, center.y, 0f);
76:             box.transform.rotation = Quaternion.Euler(0f, 0f, angle);
77:             int layer = LayerMask.NameToLayer(layerName);
78:             if (layer >= 0)
79:             {
80:                 box.layer = layer;
81:             }
82:             var renderer = box.AddComponent<SpriteRenderer>();
83:             renderer.sprite = ctx.White;
84:             renderer.color = color;
85:             renderer.sortingLayerName = sortingLayer;
86:             SizeSprite(renderer, size);
87:             if (solid)
88:             {
89:                 var collider = box.AddComponent<BoxCollider2D>();
90:                 collider.size = size;
91:                 collider.sharedMaterial = ctx.Friction;
92:                 if (oneWay)
93:                 {
94:                     collider.usedByEffector = true;
95:                     var effector = box.AddComponent<PlatformEffector2D>();
96:                     effector.useOneWay = true;
97:                 }
98:             }
99:             return box;
100:         }

102:         /// <summary>斜坡盒子：p1→p2 为上表面连线，厚度向下延伸（树枝缓坡）。</summary>
103:         internal static GameObject SlopeBox(RoomContext ctx, string name, Vector2 p1, Vector2 p2, float thickness,
104:             Color color, string sortingLayer, string layerName)
105:         {
106:             Vector2 direction = (p2 - p1).normalized;
107:             Vector2 upNormal = new Vector2(-direction.y, direction.x);
108:             if (upNormal.y < 0f)
109:             {
110:                 upNormal = -upNormal; // 保证法线朝上
111:             }
112:             Vector2 center = (p1 + p2) * 0.5f - upNormal * (thickness * 0.5f);
113:             float length = Vector2.Distance(p1, p2);
114:             float angle = Mathf.Atan2(direction.y, direction.x) * Mathf.Rad2Deg;
115:             return Box(ctx, name, center, new Vector2(length, thickness), color, sortingLayer, layerName,
116:                 solid: true, oneWay: false, angle);
117:         }

264:         /// <summary>梯子内容：视觉立杆+横档 + 攀爬 Trigger（挂到过去状态子物体，策划 11.3）。
265:         /// centerX/bottomY/topY 均为世界坐标。</summary>
266:         internal static void LadderContent(RoomContext ctx, GameObject eraChild,
267:             float centerX, float bottomY, float topY)
268:         {
269:             float height = topY - bottomY;
270:             float midY = (bottomY + topY) * 0.5f;
271:             Vector2 parentWorld = eraChild.transform.position;
272: 
273:             // 视觉立杆（无碰撞）。
274:             var bar = new GameObject("Bar");
275:             bar.transform.SetParent(eraChild.transform, false);
276:             bar.transform.localPosition = new Vector3(centerX, midY, 0f) - (Vector3)parentWorld;
277:             var barRenderer = bar.AddComponent<SpriteRenderer>();
278:             barRenderer.sprite = ctx.White;
279:             barRenderer.color = new Color(0.55f, 0.38f, 0.22f);
280:             barRenderer.sortingLayerName = "EraPlatform";
281:             SizeSprite(barRenderer, new Vector2(0.18f, height));
282: 
283:             // 横档（无碰撞，每 0.5 单位一根）。
284:             for (float y = bottomY + 0.3f; y < topY; y += 0.5f)
285:             {
286:                 var rung = new GameObject($"Rung_{y:0.0}");
287:                 rung.transform.SetParent(eraChild.transform, false);
288:                 rung.transform.localPosition = new Vector3(centerX, y, 0f) - (Vector3)parentWorld;
289:                 var rungRenderer = rung.AddComponent<SpriteRenderer>();
290:                 rungRenderer.sprite = ctx.White;
291:                 rungRenderer.color = new Color(0.7f, 0.55f, 0.35f);
292:                 rungRenderer.sortingLayerName = "EraPlatform";
293:                 SizeSprite(rungRenderer, new Vector2(0.55f, 0.07f));
294:             }
295: 
296:             // 攀爬触发区（梯顶超出平台面 0.7，便于爬出后落到相邻平台，策划 11.3）。
297:             // 触发器放独立的 Ladder 层：地面探测掩码（Ground|Temporal）不含它，玩家贴梯悬空不会被误判为落地。
298:             var zone = new GameObject("LadderZone");
299:             zone.transform.SetParent(eraChild.transform, false);
300:             zone.transform.localPosition = new Vector3(centerX, midY, 0f) - (Vector3)parentWorld;
301:             int ladderLayer = LayerMask.NameToLayer("Ladder");
302:             if (ladderLayer >= 0)
303:             {
304:                 zone.layer = ladderLayer;
305:             }
306:             var trigger = zone.AddComponent<BoxCollider2D>();
307:             trigger.isTrigger = true;
308:             trigger.size = new Vector2(0.8f, height);
309:             zone.AddComponent<LadderZone>();
310:         }

314:         internal static void SetField(UnityEngine.Object target, string field, object value)
315:         {
316:             var so = new SerializedObject(target);
317:             SerializedProperty property = so.FindProperty(field);
318:             if (property == null)
319:             {
320:                 Debug.LogWarning($"[LevelLayouts] 找不到字段 {field}（{target.GetType().Name}）。");
321:                 return;
322:             }
323:             switch (value)
324:             {
325:                 case float f: property.floatValue = f; break;
326:                 case int i: property.intValue = i; break;
327:                 case bool b: property.boolValue = b; break;
328:                 case string s: property.stringValue = s; break;
329:                 case Vector2 v: property.vector2Value = v; break;
330:             }
331:             so.ApplyModifiedPropertiesWithoutUndo();
332:         }
333: 
334:         internal static void SetObject(UnityEngine.Object target, string field, UnityEngine.Object value)
335:         {
336:             var so = new SerializedObject(target);
337:             SerializedProperty property = so.FindProperty(field);
338:             if (property != null)
339:             {
340:                 property.objectReferenceValue = value;
341:                 so.ApplyModifiedPropertiesWithoutUndo();
342:             }
343:         }

667:         private static void BoxChild(RoomContext ctx, Transform parent, string name, Vector2 center, Vector2 size,
668:             Color color, bool solid = true, float angle = 0f)
669:         {
670:             var box = new GameObject(name);
671:             box.transform.SetParent(parent, false);
672:             // center 为世界坐标：换算到父物体（年代状态子物体）的局部坐标。
673:             box.transform.localPosition = new Vector3(center.x, center.y, 0f) - parent.position;
674:             box.transform.rotation = Quaternion.Euler(0f, 0f, angle);
675:             box.layer = parent.gameObject.layer;
676:             var renderer = box.AddComponent<SpriteRenderer>();
677:             renderer.sprite = ctx.White;
678:             renderer.color = color;
679:             renderer.sortingLayerName = "EraPlatform";
680:             SizeSprite(renderer, size);
681:             if (solid)
682:             {
683:                 var collider = box.AddComponent<BoxCollider2D>();
684:                 collider.size = size;
685:                 collider.sharedMaterial = ctx.Friction;
686:             }
687:         }
688: 
689:         private static void SlopeChild(RoomContext ctx, Transform parent, string name, Vector2 p1, Vector2 p2,
690:             float thickness)
691:         {
692:             Vector2 direction = (p2 - p1).normalized;
693:             Vector2 upNormal = new Vector2(-direction.y, direction.x);
694:             if (upNormal.y < 0f)
695:             {
696:                 upNormal = -upNormal;
697:             }
698:             Vector2 center = (p1 + p2) * 0.5f - upNormal * (thickness * 0.5f);
699:             float length = Vector2.Distance(p1, p2);
700:             float angle = Mathf.Atan2(direction.y, direction.x) * Mathf.Rad2Deg;
701:             BoxChild(ctx, parent, name, center, new Vector2(length, thickness), PresentColor, solid: true, angle);
702:         }
