82:         private void Update()
83:         {

89:             if (session.State.Phase == GamePhase.Credits && !credits)
90:             {
91:                 credits = true;

93:                 panel.SetActive(true); backdrop.color = Color.black; backdrop.raycastTarget = true;
94:                 frame.gameObject.SetActive(false); speaker.gameObject.SetActive(false); body.gameObject.SetActive(false);
95:                 if (creditsBackground != null)
96:                 {
97:                     ResetActionView();
98:                     picture.texture = creditsBackground; picture.color = new Color(.55f,.55f,.55f,1);
99:                     FrameRect.anchoredPosition = Vector2.zero; FrameRect.sizeDelta = new Vector2(1920,1080);
100:                     frame.gameObject.SetActive(true);
101:                 }
102:                 actionPrompt.gameObject.SetActive(false);
103:                 next.SetActive(false); skip.SetActive(false); help.gameObject.SetActive(false); title.gameObject.SetActive(true);
104:                 title.text = "借一点明天\n\n谢谢你陪外婆走过这段回忆。";
105:             }
106:             else if (credits && session.State.Phase != GamePhase.Credits) { credits = false; Close(); }
107:         }
108:         public void Present(DialogueEntry entry, string text, bool locksControl, int page, int pages)
109:         {
110:             displayed = entry;
111:             blocking = locksControl;
112:             panel.SetActive(true);
113:             bool image = entry.illustration != null;
114:             backdrop.color = new Color(0.025f,0.022f,0.02f,(image || entry.blackBackground || entry.centeredTitle) ? 1f : 0f);
115:             // Blocking background clicks also avoids dragging the time handle underneath an illustration.
116:             backdrop.raycastTarget = locksControl;
117:             frame.gameObject.SetActive(image);
118:             // The line right after an action keeps its close-up; any later line returns to the full picture.
119:             bool keepClose = heldAction != null && heldAction.beforeEntryId == entry.entryId && heldAction.afterImage == entry.illustration;
120:             if (!keepClose) { heldAction = null; ResetActionView(); }
121:             if (image) ShowPicture(entry.illustration);
122:             actionPrompt.gameObject.SetActive(false);
123:             bool isTitle = entry.centeredTitle;
124:             title.gameObject.SetActive(isTitle); title.text = isTitle ? text : string.Empty;
125:             speaker.gameObject.SetActive(!isTitle); body.gameObject.SetActive(!isTitle);
126:             panel.transform.Find("ReadingPanel").gameObject.SetActive(!isTitle);
127:             speaker.text = entry.speakerLabel;
128:             speaker.color = entry.source == "Tape" ? new Color(0.96f,0.77f,0.43f) : entry.source == "Reality" ? new Color(0.65f,0.86f,0.94f) : Color.white;
129:             if(subtitleSkin!=null) speaker.color=entry.source=="Reality"?new Color(.1f,.25f,.29f):new Color(.3f,.17f,.07f);
130:             body.text = text;
131:             help.gameObject.SetActive(!isTitle);
132:             help.text = locksControl ? "E / 鼠标左键继续" + (pages > 1 ? "    " + page + " / " + pages : "") : string.Empty;
133:             next.SetActive(locksControl && entry.holdForConfirm && !isTitle);
134:             skip.SetActive(locksControl && !isTitle);
135:             ApplySubtitleSettings();
136:         }
137:         public void PresentAction(NarrativeAction action, float progress, string prompt, bool locksControl)
138:         {
139:             displayed = null;
140:             heldAction = action;
141:             blocking = locksControl;
142:             panel.SetActive(true);
143:             backdrop.color = new Color(0.025f,0.022f,0.02f,1f);
144:             backdrop.raycastTarget = locksControl;
145:             float t = Mathf.SmoothStep(0f, 1f, progress);
146:             Texture2D texture = action.beforeImage != null && t < 0.5f ? action.beforeImage : action.afterImage;
147:             frame.gameObject.SetActive(texture != null);
148:             if (texture != null) ShowPicture(texture);
149:             picture.rectTransform.pivot = action.focus;
150:             picture.rectTransform.localScale = Vector3.one * Mathf.Lerp(1f, Mathf.Max(1f, action.zoom), t);
151:             // Placeholder before-state until dedicated art exists: a cover over the changed area moves away.
152:             bool useCover = action.beforeImage == null && action.coverRect.width > 0f && t < 1f;
153:             cover.gameObject.SetActive(useCover);
154:             if (useCover)
155:             {
156:                 RectTransform rt = cover.rectTransform;
157:                 rt.anchorMin = action.coverRect.min; rt.anchorMax = action.coverRect.max;
158:                 rt.sizeDelta = Vector2.zero;
159:                 rt.anchoredPosition = Vector2.Scale(action.coverSlide, FrameRect.sizeDelta) * t;
160:                 Color c = action.coverColor; c.a *= 1f - Mathf.SmoothStep(0.55f, 1f, t);
161:                 cover.color = c;
162:             }
163:             title.gameObject.SetActive(false);
164:             panel.transform.Find("ReadingPanel").gameObject.SetActive(!string.IsNullOrEmpty(prompt));
165:             speaker.gameObject.SetActive(false); body.gameObject.SetActive(false);
166:             actionPrompt.gameObject.SetActive(!string.IsNullOrEmpty(prompt));
167:             actionPrompt.text = string.IsNullOrEmpty(prompt) ? string.Empty : "[E] " + prompt;
168:             actionPrompt.color = subtitleSkin != null ? new Color(.3f,.17f,.07f) : Color.white;
169:             help.gameObject.SetActive(false);
170:             next.SetActive(false);
171:             skip.SetActive(locksControl);
172:         }
173:         private void ShowPicture(Texture2D texture)
174:         {
175:             picture.texture = texture;
176:             picture.color = Color.white;
177:             FrameRect.anchoredPosition = PictureCenter;
178:             float ratio = (float)texture.width / texture.height;
179:             FrameRect.sizeDelta = ratio > PictureMax.x / PictureMax.y
180:                 ? new Vector2(PictureMax.x, PictureMax.x / ratio) : new Vector2(PictureMax.y * ratio, PictureMax.y);
181:         }
182:         private void ResetActionView()
183:         {
184:             picture.rectTransform.pivot = new Vector2(.5f,.5f);
185:             picture.rectTransform.localScale = Vector3.one;
186:             cover.gameObject.SetActive(false);
187:         }
188:         private void ApplySubtitleSettings()
189:         {
190:             if (displayed == null || displayed.centeredTitle) return;
191:             bool readableObject = displayed.source == "Note";
192:             bool enabled = !ServiceRegistry.TryGet<ISaveService>(out var save) || save.LoadSettings().subtitlesEnabled;
193:             // Object text remains readable even when spoken subtitles are disabled.
194:             body.gameObject.SetActive(enabled || readableObject);
195:             speaker.gameObject.SetActive(enabled || readableObject);
196:         }
197:         public void Close() { displayed = null; if (panel != null) panel.SetActive(false); }
198:         private T Node<T>(string name, Transform parent, Vector2 position, Vector2 size, bool stretch=false) where T : Component
199:         {
200:             var go = new GameObject(name, typeof(RectTransform)); go.transform.SetParent(parent,false);
201:             var rt = go.GetComponent<RectTransform>();
202:             if (stretch) { rt.anchorMin=Vector2.zero;rt.anchorMax=Vector2.one;rt.offsetMin=rt.offsetMax=Vector2.zero; }
203:             else { rt.anchorMin=rt.anchorMax=new Vector2(.5f,.5f);rt.anchoredPosition=position;rt.sizeDelta=size; }
204:             return go.AddComponent<T>();
205:         }
