31:         private static Sprite MakeSprite(string name,string texture,float[] rect,Vector2 pivot,float ppu)
32:         {

35:             Texture2D tex=AssetDatabase.LoadAssetAtPath<Texture2D>(texture);
36:             if(tex==null)throw new InvalidOperationException("Missing art: "+texture);

40:             float sx=(float)tex.width/sourceWidth,sy=(float)tex.height/sourceHeight;
41:             var area=new Rect(rect[0]*sx,rect[1]*sy,rect[2]*sx,rect[3]*sy);
42:             var s=Sprite.Create(tex,area,pivot,ppu*sy,0,SpriteMeshType.FullRect);s.name=name;
43:             AssetDatabase.CreateAsset(s,path);return s;
